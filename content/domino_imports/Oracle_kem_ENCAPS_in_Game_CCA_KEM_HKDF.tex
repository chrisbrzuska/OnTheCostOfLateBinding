\pcif \n{hon}_\n{rand} \pcthen\\
\pcind \n{rand} \stackrel{rand}{\sample} \bin^{\n{rand}_\n{len}}\\

 \left(\n{ctxt}, \n{ss}\right) \gets \O{\n{encaps}}(\n{pk}, \n{rand})\\

 \pcreturn \n{ctxt}\\
}
